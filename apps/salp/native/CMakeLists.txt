cmake_minimum_required(VERSION 3.14)
project(salp_rdp LANGUAGES C CXX)
find_package(Threads REQUIRED)
add_library(salp_rdp SHARED rdp_bridge.cc)
target_compile_features(salp_rdp PRIVATE cxx_std_17)
if(WIN32)
  find_package(FreeRDP 3 CONFIG REQUIRED)
  find_package(WinPR 3 CONFIG REQUIRED)
  target_compile_definitions(salp_rdp PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(salp_rdp PRIVATE /W4 /EHsc)
  target_link_libraries(salp_rdp PRIVATE freerdp winpr Threads::Threads)
  add_subdirectory(windows)
  install(TARGETS salp_rdp RUNTIME DESTINATION bin)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FREERDP REQUIRED IMPORTED_TARGET freerdp3 winpr3)
  target_compile_options(salp_rdp PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(salp_rdp PRIVATE PkgConfig::FREERDP Threads::Threads)
endif()
